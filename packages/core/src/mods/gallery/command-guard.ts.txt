import type { Register } from 'claude-code'

export const register: Register = (on, options) => {
  const rules: RegExp[] = []
  const unreadable: string[] = []

  for (const source of options.blocked as readonly string[]) {
    if (source.trim() === '') {
      continue
    }
    try {
      rules.push(new RegExp(source, 'i'))
    } catch {
      unreadable.push(source)
    }
  }

  on('session.start', ($, e, next) => {
    if (unreadable.length > 0) {
      $.ui.log(`Command guard skipped ${unreadable.length} pattern(s) that are not regular expressions: ${unreadable.join('   ')}`)
    }

    return next(e)
  })

  on('tool.call', { tool: 'Bash' }, ($, e, next) => {
    const rule = rules.find(one => one.test(e.command))

    return rule === undefined ? next(e) : { deny: `${String(options.message)} (Matched: ${rule.source})` }
  })
}
