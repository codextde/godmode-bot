import type { Register } from 'claude-code'

// A shell command reaches a computer through Bash, and through Godmode's tools for servers and virtual machines
// (mcp__ssh__shell, mcp__vm__shell).
function shellCommand(e: { tool: string }): string | undefined {
  const command = (e as { command?: unknown }).command
  const isShell = e.tool === 'Bash' || /^mcp__.+__shell$/.test(e.tool)

  return isShell && typeof command === 'string' ? command : undefined
}

export const register: Register = (on, options) => {
  const rules: RegExp[] = []
  const unreadable: string[] = []

  for (const source of options.blocked as readonly string[]) {
    if (source.trim() === '') {
      continue
    }
    try {
      rules.push(new RegExp(source, 'i'))
    } catch {
      unreadable.push(source)
    }
  }

  on('session.start', ($, e, next) => {
    if (unreadable.length > 0) {
      $.ui.log(`Command guard skipped ${unreadable.length} pattern(s) that are not regular expressions: ${unreadable.join('   ')}`)
    }

    return next(e)
  })

  on('tool.call', ($, e, next) => {
    const command = shellCommand(e)

    return command !== undefined && rules.some(rule => rule.test(command)) ? { deny: String(options.message) } : next(e)
  })
}
