import type { Register } from 'claude-code'

const MASK = '[secret removed]'

const KNOWN: readonly RegExp[] = [
  /-----BEGIN [A-Z ]*PRIVATE KEY-----[\s\S]*?-----END [A-Z ]*PRIVATE KEY-----/g,
  /\bAKIA[0-9A-Z]{16}\b/g,
  /\bgh[pousr]_[A-Za-z0-9]{36,}\b/g,
  /\bgithub_pat_[A-Za-z0-9_]{40,}\b/g,
  /\bsk-[A-Za-z0-9_-]{20,}/g,
  /\bxox[abprs]-[A-Za-z0-9-]{10,}/g,
  /\b[prs]k_(?:live|test)_[A-Za-z0-9]{16,}\b/g,
  /\bAIza[0-9A-Za-z_-]{35}\b/g,
  /\beyJ[A-Za-z0-9_-]{10,}\.eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}/g,
]

// What comes before the secret stays: "Bearer ", and a URL up to the password.
const AFTER: readonly RegExp[] = [/\b(Bearer\s+)[A-Za-z0-9._~+/=-]{20,}/g, /(:\/\/[^/\s:@]+:)[^@\s/]{3,}(?=@)/g]

const NAME = String.raw`(?:[a-z0-9]+[_-])*(?:password|passwd|secret|token|api[_-]?key|access[_-]?key|private[_-]?key)`
// The value after a name that sounds like a secret counts when it is written out: in quotes without a space, or a
// bare word with a digit or a symbol in it. A call, a property or a plain identifier (generateToken(user),
// process.env.TOKEN, undefined) is code — masking it would hand the agent a file that isn't the one on disk.
const QUOTED = new RegExp(String.raw`\b(${NAME}["']?\s*[=:]\s*["'])([^"'\s]{8,})(?=["'])`, 'gi')
const BARE = new RegExp(String.raw`\b(${NAME}["']?\s*[=:]\s*)([A-Za-z0-9_+/=@%~!#^*-]{8,})(?=$|[\s,;)}\]])`, 'gim')

export const register: Register = (on, options) => {
  const patterns: RegExp[] = [...KNOWN]
  const unreadable: string[] = []

  for (const source of options.extra as readonly string[]) {
    if (source.trim() === '') {
      continue
    }
    try {
      patterns.push(new RegExp(source, 'g'))
    } catch {
      unreadable.push(source)
    }
  }

  on('session.start', ($, e, next) => {
    if (unreadable.length > 0) {
      $.ui.log(`Secret scrubber skipped ${unreadable.length} pattern(s) that are not regular expressions: ${unreadable.join('   ')}`)
    }

    return next(e)
  })

  on('session.append', { door: 'tool-result' }, ($, e, next) => {
    let masked = 0

    const scrub = (text: string): string => {
      let out = text
      for (const pattern of patterns) {
        out = out.replace(pattern, () => {
          masked += 1

          return MASK
        })
      }
      for (const pattern of AFTER) {
        out = out.replace(pattern, (whole, before: string) => {
          masked += 1

          return before + MASK
        })
      }
      if (options.assignments === true) {
        const assigned = (whole: string, name: string, value: string) => {
          if (value.includes('[secret')) {
            return whole
          }
          masked += 1

          return name + MASK
        }
        out = out.replace(QUOTED, assigned)
        out = out.replace(BARE, (whole, name: string, value: string) =>
          /[^A-Za-z_]/.test(value) ? assigned(whole, name, value) : whole,
        )
      }

      return out
    }

    const content = e.message.content.map(block => {
      if (block.type !== 'tool_result') {
        return block
      }
      if (typeof block.content === 'string') {
        return { ...block, content: scrub(block.content) }
      }
      if (Array.isArray(block.content)) {
        return {
          ...block,
          content: block.content.map(part => (part.type === 'text' ? { ...part, text: scrub(part.text) } : part)),
        }
      }

      return block
    })

    if (masked === 0) {
      return next(e)
    }
    if (options.notify === true) {
      const tool = e.origin.kind === 'tool' ? e.origin.tool : 'a tool'
      $.ui.log(`Masked ${masked} secret${masked === 1 ? '' : 's'} in the output of ${tool}`)
    }

    return next({ ...e, message: { ...e.message, content } })
  })
}
