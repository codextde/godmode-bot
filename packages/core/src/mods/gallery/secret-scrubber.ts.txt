import type { Register } from 'claude-code'

const MASK = '[secret removed]'

const KNOWN: readonly RegExp[] = [
  /-----BEGIN [A-Z ]*PRIVATE KEY-----[\s\S]*?-----END [A-Z ]*PRIVATE KEY-----/g,
  /\bAKIA[0-9A-Z]{16}\b/g,
  /\bgh[pousr]_[A-Za-z0-9]{36,}\b/g,
  /\bgithub_pat_[A-Za-z0-9_]{40,}\b/g,
  /\bsk-[A-Za-z0-9_-]{20,}/g,
  /\bxox[abprs]-[A-Za-z0-9-]{10,}/g,
  /\b[prs]k_(?:live|test)_[A-Za-z0-9]{16,}\b/g,
  /\bAIza[0-9A-Za-z_-]{35}\b/g,
  /\beyJ[A-Za-z0-9_-]{10,}\.eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}/g,
]

const ASSIGNED =
  /\b((?:[a-z0-9]+[_-])*(?:password|passwd|secret|token|api[_-]?key|access[_-]?key|private[_-]?key)["']?\s*[=:]\s*["']?)([^\s"',;]{8,})/gi

export const register: Register = (on, options) => {
  const patterns: RegExp[] = [...KNOWN]

  for (const source of options.extra as readonly string[]) {
    if (source.trim() === '') {
      continue
    }
    try {
      patterns.push(new RegExp(source, 'g'))
    } catch {
      // A pattern that is no regular expression masks nothing.
    }
  }

  on('session.append', { door: 'tool-result' }, ($, e, next) => {
    let masked = 0

    const scrub = (text: string): string => {
      let out = text
      for (const pattern of patterns) {
        out = out.replace(pattern, () => {
          masked += 1

          return MASK
        })
      }
      if (options.assignments === true) {
        out = out.replace(ASSIGNED, (whole, name: string, value: string) => {
          if (value === MASK || value.startsWith('[secret')) {
            return whole
          }
          masked += 1

          return name + MASK
        })
      }

      return out
    }

    const content = e.message.content.map(block => {
      if (block.type !== 'tool_result') {
        return block
      }
      if (typeof block.content === 'string') {
        return { ...block, content: scrub(block.content) }
      }
      if (Array.isArray(block.content)) {
        return {
          ...block,
          content: block.content.map(part => (part.type === 'text' ? { ...part, text: scrub(part.text) } : part)),
        }
      }

      return block
    })

    if (masked === 0) {
      return next(e)
    }
    if (options.notify === true) {
      const tool = e.origin.kind === 'tool' ? e.origin.tool : 'a tool'
      $.ui.log(`Masked ${masked} secret${masked === 1 ? '' : 's'} in the output of ${tool}`)
    }

    return next({ ...e, message: { ...e.message, content } })
  })
}
